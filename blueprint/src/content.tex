\chapter{Introduction}

This blueprint formalizes a combinatorial description of curve neighborhoods
for the affine Weyl group of type $\widetilde A_1$, identified with the
infinite dihedral group $D_\infty$.

The strategy is:
\begin{enumerate}
  \item define chains and degree constraints,
  \item characterize the set $A_d(u)$ and its maximal elements,
  \item identify curve neighborhoods with translates of those maxima.
\end{enumerate}

All key results are formalized in Lean in
\texttt{Dihedral/alternatingword.lean}, \texttt{Dihedral/Degree'.lean}, and
\texttt{Dihedral/Neib'.lean}.

\chapter{Algebraic Preliminaries}

\begin{theorem}\label{gen_by_sr0_sr1}
  \lean{gen_by_sr0_sr1}
  \leanok
  The group $D_\infty$ is generated by the two simple reflections $sr(0)$ and
  $sr(1)$.
\end{theorem}

\begin{definition}\label{length_def}
  \leanok
  Let $\ell : D_\infty \to \mathbb N$ be the Coxeter length.
\end{definition}

\begin{theorem}\label{length_sr_abs}
  \uses{length_def}
  \lean{length_sr_abs}
  \leanok
  For every $k \in \mathbb Z$,
  \[
    \ell(sr\,k) = |2k-1|.
  \]
\end{theorem}

\begin{theorem}\label{length_eq}
  \uses{length_def}
  \lean{length_eq}
  \leanok
  For every $g \in D_\infty$,
  \[
    \ell(g)=\operatorname{length}(\mathrm{reducedWord}(g)).
  \]
\end{theorem}

\chapter{Degrees, Chains, and Order}

\begin{definition}\label{degree_def}
  \leanok
  A degree is a pair $(a,b)\in\mathbb N^2$ with product order.
\end{definition}

\begin{definition}\label{root_def}
  \leanok
  A root is a pair $(a,b)$ such that $a=b+1$ or $b=a+1$.
\end{definition}

\begin{definition}\label{chain_def}
  \leanok
  \texttt{HasChain u v d} means that $v$ is reached from $u$ by a finite chain
  of edge moves whose total degree is $d$.
\end{definition}

\begin{theorem}\label{length_mul_eq_add_or_sub}
  \uses{length_def}
  \lean{length_mul_eq_add_or_sub}
  \leanok
  If $\ell(u)\le \ell(v)$, then
  \[
    \ell(uv)=\ell(u)+\ell(v)
    \quad\text{or}\quad
    \ell(uv)=\ell(v)-\ell(u).
  \]
\end{theorem}

\begin{theorem}\label{lt_iff_length_lt}
  \uses{chain_def, length_def}
  \lean{lt_iff_length_lt}
  \leanok
  For vertices $u,v$,
  \[
    u < v \iff \ell(u) < \ell(v).
  \]
\end{theorem}

\chapter{The Set \texorpdfstring{$A_d(u)$}{Ad(u)}}

\begin{definition}\label{ad_def}
  \leanok
  For $u\in D_\infty$ and degree bound $d$,
  \[
    A_d(u):=\{v\mid \ell(uv)=\ell(u)+\ell(v),\ \varphi(v)\le d\}.
  \]
  In Lean this is \texttt{Ad u d}.
\end{definition}

\begin{definition}\label{maximal_def}
  \leanok
  \texttt{IsMaximalIn v S} means $v\in S$ and every element of $S$ above $v$ is
  equal to $v$.
\end{definition}

\begin{theorem}\label{exists_unique_max_ad_ne_one}
  \uses{ad_def, maximal_def}
  \lean{exists_unique_max_ad_ne_one}
  \leanok
  If $u\ne 1$, then $A_d(u)$ has a unique maximal element.
\end{theorem}

\begin{theorem}\label{exists_unique_max_ad_one_neq}
  \uses{ad_def, maximal_def}
  \lean{exists_unique_max_ad_one_neq}
  \leanok
  If $d.a\ne d.b$, then $A_d(1)$ has a unique maximal element.
\end{theorem}

\begin{theorem}\label{max_ad_one_diag_iff_r_or_neg_r}
  \uses{ad_def, maximal_def}
  \lean{max_ad_one_diag_iff_r_or_neg_r}
  \leanok
  For diagonal degree $(a,a)$, maximal elements of $A_{(a,a)}(1)$ are exactly
  \[
    r(a),\ r(-a).
  \]
\end{theorem}

\chapter{Curve Neighborhood at the Identity}

\begin{definition}\label{reachable_def}
  \leanok
  \[
    \mathrm{ReachableSet}(u,d):=\{v\mid \exists d'\le d,\ \texttt{HasChain }u\ v\ d'\}.
  \]
\end{definition}

\begin{definition}\label{curve_nbhd_def}
  \uses{reachable_def, maximal_def}
  \leanok
  \[
    \mathrm{CurveNeighborhood}(u,d)
    :=\{v\mid \texttt{IsMaximalIn }v\ (\mathrm{ReachableSet}(u,d))\}.
  \]
\end{definition}

\begin{theorem}\label{curve_nbhd_one_eq_max_ad_one}
  \uses{curve_nbhd_def, ad_def}
  \lean{curve_nbhd_one_eq_max_ad_one}
  \leanok
  \[
    \mathrm{CurveNeighborhood}(1,d)
    = \{v\mid \texttt{IsMaximalIn }v\ (A_d(1))\}.
  \]
\end{theorem}

\begin{theorem}\label{curve_nbhd_one_diag}
  \uses{curve_nbhd_one_eq_max_ad_one, max_ad_one_diag_iff_r_or_neg_r}
  \lean{curve_nbhd_one_diag}
  \leanok
  If $d.a=d.b$, then
  \[
    \mathrm{CurveNeighborhood}(1,d)=\{\texttt{s0s1\_pow }d.a,\ \texttt{s1s0\_pow }d.a\}.
  \]
\end{theorem}

\begin{theorem}\label{curve_nbhd_one_offdiag}
  \uses{curve_nbhd_one_eq_max_ad_one, exists_unique_max_ad_one_neq}
  \lean{curve_nbhd_one_offdiag}
  \leanok
  If $d.a\ne d.b$, then
  \[
    \mathrm{CurveNeighborhood}(1,d)=\{\texttt{s\_alpha\_d }d\}.
  \]
\end{theorem}

\begin{theorem}\label{curve_nbhd_one}
  \uses{curve_nbhd_one_diag, curve_nbhd_one_offdiag}
  \lean{curve_nbhd_one}
  \leanok
  Complete classification:
  \[
    \mathrm{CurveNeighborhood}(1,d)=
    \begin{cases}
      \{\texttt{s0s1\_pow }d.a,\ \texttt{s1s0\_pow }d.a\}, & d.a=d.b,\\
      \{\texttt{s\_alpha\_d }d\}, & d.a\ne d.b.
    \end{cases}
  \]
\end{theorem}

\chapter{General Base Point and Main Result}

\begin{lemma}\label{max_ad_ends_s0}
  \lean{max_ad_ends_s0}
  \leanok
  Under the technical hypothesis \texttt{ends\_in\_s0 u}, one gets explicit
  maximal elements of $A_d(u)$ by case split on $d.a$ and $d.b$.
\end{lemma}

\begin{theorem}\label{deg_inv_mul_le_curve_nbhd}
  \uses{curve_nbhd_def}
  \lean{deg_inv_mul_le_curve_nbhd}
  \leanok
  If $v\in \mathrm{CurveNeighborhood}(u,d)$, then
  \[
    \varphi(u^{-1}v)\le d.
  \]
\end{theorem}

\begin{theorem}\label{inv_mul_mem_ad_curve_nbhd}
  \uses{deg_inv_mul_le_curve_nbhd, ad_def}
  \lean{inv_mul_mem_ad_curve_nbhd}
  \leanok
  If $v\in \mathrm{CurveNeighborhood}(u,d)$, then
  \[
    u^{-1}v\in A_d(u).
  \]
\end{theorem}

\begin{theorem}\label{curve_nbhd_eq_mul_max_ad}
  \uses{curve_nbhd_def, ad_def, maximal_def}
  \lean{curve_nbhd_eq_mul_max_ad}
  \leanok
  For every $u\in D_\infty$ and every degree bound $d$,
  \[
    \mathrm{CurveNeighborhood}(u,d)
    =
    \{v\mid \exists w,\ \texttt{IsMaximalIn }w\ (A_d(u))\ \wedge\ v=u\cdot w\}.
  \]
\end{theorem}

\begin{proof}
  \leanok
  The forward direction uses \ref{inv_mul_mem_ad_curve_nbhd} to put
  $u^{-1}v$ in $A_d(u)$, then maximality arguments in the length order.
  The reverse direction lifts maximal elements of $A_d(u)$ to reachable maximal
  points via left multiplication.
\end{proof}
  Hence $\ell(u \cdot w) = \ell(u) + \ell(w)$, so $w \in A_d(u)$.
